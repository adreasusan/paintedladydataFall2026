const CENTRAL_SHEET_ID = '16hI-nNTciADPBjI6O0lrUrAxIbYRC6sXHyzU26jb5AE'; const PUPAL_TAB = 'Pupal development'; const FOLDER_ID = '1VAeDuDnkAA8OmAcEqd3yjcQT74QQ_noC'; function doPost(e) { try { const data = JSON.parse(e.postData.contents); return writePupal(data); } catch(err) { return ContentService .createTextOutput(JSON.stringify({success: false, error: err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function writePupal(data) { // 1 — Per-container workbook (date-named tabs, metadata + individual measurements) const folder = DriveApp.getFolderById(FOLDER_ID); const fileName = data.container + '_pupal'; let ss; const existing = folder.getFilesByName(fileName); if (existing.hasNext()) { ss = SpreadsheetApp.open(existing.next()); } else { ss = SpreadsheetApp.create(fileName); DriveApp.getFileById(ss.getId()).moveTo(folder); const defaultSheet = ss.getSheetByName('Sheet1'); if (defaultSheet) ss.deleteSheet(defaultSheet); } const tabName = data.date; let sheet = ss.getSheetByName(tabName); if (!sheet) sheet = ss.insertSheet(tabName); // Metadata block (rows 1-3) sheet.getRange(1,1).setValue("Today's Date:"); sheet.getRange(1,2).setValue(data.date); sheet.getRange(1,3).setValue("Diet:"); sheet.getRange(1,4).setValue(data.diet); sheet.getRange(1,5).setValue("Temperature:"); sheet.getRange(1,6).setValue(data.temp + '°C'); sheet.getRange(2,1).setValue("Total Pupae Alive:"); sheet.getRange(2,2).setValue(data.alive); sheet.getRange(2,3).setValue("Pupal Deaths:"); sheet.getRange(2,4).setValue(data.deaths); sheet.getRange(2,5).setValue("Observers:"); sheet.getRange(2,6).setValue(data.observers); sheet.getRange(3,1).setValue("Container:"); sheet.getRange(3,2).setValue(data.container); sheet.getRange(3,5).setValue("Campus:"); sheet.getRange(3,6).setValue(data.campus); // Header row (row 4) sheet.getRange(4,1).setValue("Pupa Length (mm)"); sheet.getRange(4,2).setValue("Pupa Width (mm)"); sheet.getRange(4,1,1,2).setFontWeight('bold').setBackground('#d4e8d8'); // Individual measurements (rows 5+) if (data.pupae && data.pupae.length > 0) { data.pupae.forEach(function(p, i) { if (p.length || p.width) { sheet.getRange(5 + i, 1).setValue(p.length || ''); sheet.getRange(5 + i, 2).setValue(p.width || ''); } }); } // 2 — Central sheet summary row const HEADERS = ['Date','Campus','Temperature','Diet','Cup_ID', 'Total_Pupae_Alive','Pupal_Deaths','Avg_Pupa_Length_mm','Avg_Pupa_Width_mm','Observer','Notes']; const ROW = [data.date, data.campus, data.temp, data.diet, data.container, data.alive, data.deaths, data.avgLen, data.avgWid, data.observers, data.notes]; const central = SpreadsheetApp.openById(CENTRAL_SHEET_ID); let cSheet = central.getSheetByName(PUPAL_TAB); if (!cSheet) cSheet = central.insertSheet(PUPAL_TAB); if (cSheet.getLastRow() === 0) { cSheet.appendRow(HEADERS); cSheet.getRange(1,1,1,HEADERS.length).setFontWeight('bold').setBackground('#d4e8d8'); } cSheet.appendRow(ROW); return ContentService .createTextOutput(JSON.stringify({success: true})) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { return ContentService .createTextOutput(JSON.stringify({status: 'ok'})) .setMimeType(ContentService.MimeType.JSON); }
